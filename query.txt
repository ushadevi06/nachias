-- 1. Credit Notes: Reference Number
ALTER TABLE `credit_notes`
ADD COLUMN `reference_no` VARCHAR(100) NULL AFTER `note_date`;


-- 2. Settings: Refinishing Amount
ALTER TABLE `settings`
ADD COLUMN `refinishing_amount` DECIMAL(12,2) NOT NULL DEFAULT 0
AFTER `terms_and_conditions`;


-- 3. Sales Invoices: GST Filing Details
ALTER TABLE `sales_invoices`
ADD COLUMN `is_gst_filed` TINYINT(1) NOT NULL DEFAULT 0 AFTER `einvoice_status`,
ADD COLUMN `gst_filed_at` TIMESTAMP NULL AFTER `is_gst_filed`,
ADD COLUMN `gst_filed_by` BIGINT UNSIGNED NULL AFTER `gst_filed_at`;


-- 4. Fix SO-0360 (ID 379) header total_qty mismatch with items
UPDATE `sales_orders`
SET `total_qty` = 17.00, `sub_total_qty` = 17.00
WHERE `id` = 379 AND `so_no` = 'SO-0360';


-- 5. Attendances: Reason for Manual Edit
ALTER TABLE `attendances`
ADD COLUMN `reason` TEXT NULL AFTER `manual_updated_at`;


-- 6. Leave Types Master Table
CREATE TABLE IF NOT EXISTS `leave_types` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `leave_type` VARCHAR(100) NOT NULL,
  `status` ENUM('Active', 'Inactive') NOT NULL DEFAULT 'Active',
  `created_by` BIGINT UNSIGNED NULL,
  `updated_by` BIGINT UNSIGNED NULL,
  `created_at` TIMESTAMP NULL,
  `updated_at` TIMESTAMP NULL,
  `deleted_at` TIMESTAMP NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

INSERT INTO `leave_types` (`leave_type`, `status`, `created_at`, `updated_at`) VALUES
('Casual Leave', 'Active', NOW(), NOW()),
('Sick Leave', 'Active', NOW(), NOW()),
('Paid Leave', 'Active', NOW(), NOW()),
('Compensatory Off', 'Active', NOW(), NOW());


-- 7. Leaves: Attachment File Upload
ALTER TABLE `leaves`
ADD COLUMN `attachment` VARCHAR(255) NULL AFTER `reason`;


-- 8. Leaves: Date Range and No of Days for Common Record
ALTER TABLE `leaves`
ADD COLUMN `start_date` DATE NULL AFTER `leave_type`,
ADD COLUMN `end_date` DATE NULL AFTER `start_date`,
ADD COLUMN `no_of_days` DECIMAL(5,1) NOT NULL DEFAULT 1.0 AFTER `end_date`;

UPDATE `leaves` 
SET `start_date` = COALESCE(`start_date`, `leave_date`),
    `end_date` = COALESCE(`end_date`, `leave_date`),
    `no_of_days` = COALESCE(`no_of_days`, 1.0)
WHERE `start_date` IS NULL;


-- 9. Attendances: Overtime Status (Pending, Approved, Not Approved)
ALTER TABLE `attendances`
ADD COLUMN `ot_status` VARCHAR(50) NOT NULL DEFAULT 'Pending' AFTER `status`;
