-- 1. Credit Notes: Reference Number
ALTER TABLE `credit_notes`
ADD COLUMN `reference_no` VARCHAR(100) NULL AFTER `note_date`;


-- 2. Settings: Refinishing Amount
ALTER TABLE `settings`
ADD COLUMN `refinishing_amount` DECIMAL(12,2) NOT NULL DEFAULT 0
AFTER `terms_and_conditions`;


-- 3. Sales Invoices: GST Filing Details
ALTER TABLE `sales_invoices`
ADD COLUMN `is_gst_filed` TINYINT(1) NOT NULL DEFAULT 0 AFTER `einvoice_status`,
ADD COLUMN `gst_filed_at` TIMESTAMP NULL AFTER `is_gst_filed`,
ADD COLUMN `gst_filed_by` BIGINT UNSIGNED NULL AFTER `gst_filed_at`;


-- 4. Fix SO-0360 (ID 379) header total_qty mismatch with items
UPDATE `sales_orders`
SET `total_qty` = 17.00, `sub_total_qty` = 17.00
WHERE `id` = 379 AND `so_no` = 'SO-0360';
