UPDATE stock_entry_items 
SET brand_id = (
    SELECT id 
    FROM brands 
    WHERE stock_entry_items.art_no LIKE CONCAT(brands.code, '%') 
       OR stock_entry_items.finished_item_code LIKE CONCAT(brands.code, '%')
    ORDER BY LENGTH(brands.code) DESC
    LIMIT 1
)
WHERE stock_type = 'finished_goods' 
AND brand_id IS NULL;
